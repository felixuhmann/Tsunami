function(tsunami_plugin target source id product code core)
  set(generated "${CMAKE_CURRENT_BINARY_DIR}/${target}-generated")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_metadata.py"
    "${source}/parameters.json" "${generated}/Generated.hpp" --id "${id}" --product "${product}" --core "${core}"
    COMMAND_ERROR_IS_FATAL ANY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}/parameters.json" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_metadata.py")
  juce_add_plugin(${target} COMPANY_NAME "Tsunami" BUNDLE_ID "org.tsunami.${id}"
    PLUGIN_MANUFACTURER_CODE Tsun PLUGIN_CODE "${code}" PRODUCT_NAME "${product}"
    FORMATS VST3 Standalone IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx)
  target_sources(${target} PRIVATE shared/Plugin.cpp ${ARGN}) # ARGN: optional plugin-local .cpp core sources
  target_include_directories(${target} PUBLIC shared "${source}" "${generated}")
  target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(${target} PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
  clap_juce_extensions_plugin(TARGET ${target} CLAP_ID "org.tsunami.${id}" CLAP_FEATURES audio-effect stereo)
endfunction()

set(TSUNAMI_PLUGIN_ID "" CACHE STRING "Target id: any plugins/<id> folder under TSUNAMI_PLUGIN_SOURCE_ROOT with the contract files")

  if(NOT TSUNAMI_PLUGIN_SOURCE_ROOT)
    set(TSUNAMI_PLUGIN_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
  file(TO_CMAKE_PATH "${TSUNAMI_PLUGIN_SOURCE_ROOT}" TSUNAMI_PLUGIN_SOURCE_ROOT) # native Windows roots
  tsunami_resolve_plugin("${TSUNAMI_PLUGIN_SOURCE_ROOT}" "${TSUNAMI_PLUGIN_ID}" TSUNAMI_PLUGIN_DIR)

    include("${TSUNAMI_PLUGIN_DIR}/tsunami.cmake")

    tsunami_plugin(TsunamiStereo "${TSUNAMI_PLUGIN_DIR}" s1 "Tsunami Stereo" TsS1 "tsunami::s1::Processor")

    add_test(NAME s1.editor COMMAND s1_editor_test)
    add_executable(s1_editor_rt_test "${TSUNAMI_PLUGIN_SOURCE_ROOT}/tests/s1/editor_rt_test.cpp")
    target_link_libraries(s1_editor_rt_test PRIVATE TsunamiStereo juce::juce_audio_utils)

  else()

  endif()
endif()

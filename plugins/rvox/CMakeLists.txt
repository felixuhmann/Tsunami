cmake_minimum_required(VERSION 3.22)
project(TsunamiRVoxCore LANGUAGES CXX)
include(CTest)
add_library(tsunami_rvox INTERFACE)
target_compile_features(tsunami_rvox INTERFACE cxx_std_17)
target_include_directories(tsunami_rvox INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
foreach(test IN ITEMS rvox gate process safety state ceiling)
  add_executable(rvox_test_${test} "../../tests/rvox/test_${test}.cpp")
  target_link_libraries(rvox_test_${test} PRIVATE tsunami_rvox)
  add_test(NAME rvox.${test} COMMAND rvox_test_${test})
endforeach()
add_executable(rvox_render "../../tests/rvox/render.cpp")
target_link_libraries(rvox_render PRIVATE tsunami_rvox)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME rvox.render-validation COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/rvox/test_render.py" "$<TARGET_FILE:rvox_render>")
endif()

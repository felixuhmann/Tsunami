cmake_minimum_required(VERSION 3.16)
project(tsunami_hdelay LANGUAGES CXX)
enable_testing()
add_library(tsunami_hdelay INTERFACE)
target_include_directories(tsunami_hdelay INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(tsunami_hdelay INTERFACE cxx_std_17)
add_executable(hdelay_test ../../tests/hdelay/test_processor.cpp)
target_link_libraries(hdelay_test PRIVATE tsunami_hdelay)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(hdelay_test PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_test(NAME hdelay_processor COMMAND hdelay_test)
add_executable(hdelay_render ../../tests/hdelay/render.cpp)
target_link_libraries(hdelay_render PRIVATE tsunami_hdelay)

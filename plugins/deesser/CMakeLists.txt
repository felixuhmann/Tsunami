cmake_minimum_required(VERSION 3.16)
project(tsunami_deesser LANGUAGES CXX)
enable_testing()
add_library(tsunami_deesser Processor.cpp)
target_compile_features(tsunami_deesser PUBLIC cxx_std_17)
target_include_directories(tsunami_deesser PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(tsunami_deesser PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
add_executable(deesser_tests ../../tests/deesser/test_processor.cpp)
target_link_libraries(deesser_tests PRIVATE tsunami_deesser)
add_test(NAME deesser_processor COMMAND deesser_tests)
add_executable(deesser_render ../../tests/deesser/render.cpp)
target_link_libraries(deesser_render PRIVATE tsunami_deesser)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME deesser_metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/deesser/test_metadata.py)
endif()
add_executable(deesser_automation_tests ../../tests/deesser/test_automation.cpp)
target_link_libraries(deesser_automation_tests PRIVATE tsunami_deesser)
add_test(NAME deesser_automation COMMAND deesser_automation_tests)
add_executable(deesser_editor_model_tests ../../tests/deesser/test_editor_model.cpp)
target_link_libraries(deesser_editor_model_tests PRIVATE tsunami_deesser)
add_test(NAME deesser_editor_model COMMAND deesser_editor_model_tests ${CMAKE_CURRENT_SOURCE_DIR}/parameters.json)
add_executable(deesser_splice_validity ../../tests/deesser/splice_validity.cpp)
target_link_libraries(deesser_splice_validity PRIVATE tsunami_deesser)

cmake_minimum_required(VERSION 3.16)
project(tsunami_s1 LANGUAGES CXX)
enable_testing()
add_library(tsunami_s1 INTERFACE)
target_include_directories(tsunami_s1 INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(tsunami_s1 INTERFACE cxx_std_17)
add_executable(s1_test ../../tests/s1/processor_test.cpp)
target_link_libraries(s1_test PRIVATE tsunami_s1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(s1_test PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_test(NAME s1_processor COMMAND s1_test)
add_executable(s1_safety_test ../../tests/s1/safety_test.cpp)
target_link_libraries(s1_safety_test PRIVATE tsunami_s1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(s1_safety_test PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_test(NAME s1_safety COMMAND s1_safety_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME s1_metadata COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/s1" -p test_metadata.py -v)
add_executable(s1_render ../../tests/s1/render.cpp)
target_link_libraries(s1_render PRIVATE tsunami_s1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(s1_render PRIVATE -ffp-contract=off)
endif()

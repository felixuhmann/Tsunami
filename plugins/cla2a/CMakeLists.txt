cmake_minimum_required(VERSION 3.16)
project(tsunami_cla2a LANGUAGES CXX)
enable_testing()
add_library(tsunami_cla2a INTERFACE)
target_include_directories(tsunami_cla2a INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(tsunami_cla2a INTERFACE cxx_std_17)
foreach(name memory processor mapping kernel)
  add_executable(cla2a_test_${name} ../../tests/cla2a/test_${name}.cpp)
  target_link_libraries(cla2a_test_${name} PRIVATE tsunami_cla2a)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cla2a_test_${name} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
  add_test(NAME cla2a_${name} COMMAND cla2a_test_${name})
endforeach()
add_executable(cla2a_reference_driver ../../tests/cla2a/reference_driver.cpp)
target_link_libraries(cla2a_reference_driver PRIVATE tsunami_cla2a)

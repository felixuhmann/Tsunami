cmake_minimum_required(VERSION 3.16)
project(tsunami_ns1_dsp LANGUAGES CXX)
enable_testing()
add_library(tsunami_ns1 STATIC detector.cpp mono.cpp design.cpp)
target_include_directories(tsunami_ns1 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(tsunami_ns1 PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(tsunami_ns1 PRIVATE /fp:strict /W4)
else()
  target_compile_options(tsunami_ns1 PRIVATE -ffp-contract=off -Wall -Wextra -Wpedantic)
endif()
foreach(t detector mono design initialized processor)
  add_executable(ns1_${t}_test ../../tests/ns1/test_${t}.cpp)
  target_link_libraries(ns1_${t}_test PRIVATE tsunami_ns1)
  if(NOT MSVC)
    target_compile_options(ns1_${t}_test PRIVATE -ffp-contract=off -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME ns1_${t} COMMAND ns1_${t}_test)
endforeach()
add_executable(ns1_render ../../tests/ns1/render.cpp)
target_link_libraries(ns1_render PRIVATE tsunami_ns1)
if(NOT MSVC)
  target_compile_options(ns1_render PRIVATE -ffp-contract=off)
endif()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME ns1_metadata COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/ns1" -p test_metadata.py -v)
endif()

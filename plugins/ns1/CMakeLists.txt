cmake_minimum_required(VERSION 3.16)
project(tsunami_ns1_dsp LANGUAGES CXX)
enable_testing()
add_library(tsunami_ns1 STATIC detector.cpp)
target_include_directories(tsunami_ns1 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(tsunami_ns1 PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(tsunami_ns1 PRIVATE /fp:strict /W4)
else()
  target_compile_options(tsunami_ns1 PRIVATE -ffp-contract=off -Wall -Wextra -Wpedantic)
endif()
add_executable(ns1_detector_test ../../tests/ns1/test_detector.cpp)
target_link_libraries(ns1_detector_test PRIVATE tsunami_ns1)
add_test(NAME ns1_detector COMMAND ns1_detector_test)
